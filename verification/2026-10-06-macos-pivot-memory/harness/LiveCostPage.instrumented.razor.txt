@code {
    [SupplyParameterFromQuery(Name = "b")] public int Inner { get; set; } = 10;
    [SupplyParameterFromQuery(Name = "records")] public int Records { get; set; }
    [SupplyParameterFromQuery(Name = "outer")] public int? OuterParameter { get; set; }
    private int Outer => OuterParameter ?? 1000;
    [SupplyParameterFromQuery(Name = "batch")] public int? BatchParameter { get; set; }
    private int BatchSize => BatchParameter ?? 1000;
    [SupplyParameterFromQuery(Name = "gc")] public bool ForceGc { get; set; }
    [SupplyParameterFromQuery(Name = "intervention")] public string? Intervention { get; set; }
    private sealed record Row(int Id, string A, string B, decimal Value);
    private Row[] _rows = [];
    private readonly PivotFields<Row> _fields = PivotFields.Of<Row>().Key("Id", r => r.Id).Text("A", r => r.A).Text("B", r => r.B).Number("Value", r => r.Value);
    private SnapshotPivotSource? _source;
    private ExPivot? _pivot;
    private readonly PivotLayout _layout = new() { Rows = [new("A"), new("B")], Values = [new("Value")] };
    private readonly PivotCaps _caps = new() { MaxLeaves = 1_000_000, MaxRows = 1_000_000 };
    private string _status = "ready";
    private int _batch;
    protected override void OnInitialized()
    {
        _rows = Enumerable.Range(0, Records > 0 ? Records : Outer * Inner).Select(i => new Row(i, $"A{(i / Inner) % Outer:00000}", $"B{i % Inner:0000}", i)).ToArray();
        _source = PivotSource.From(_fields.Build(_rows), _fields.Fields);
    }
    private void Tick()
    {
        var changed = Enumerable.Range(0, BatchSize).Select(i => { var j = i * Inner; return _rows[j] = _rows[j] with { Value = _rows[j].Value + 1 }; }).ToArray();
        var batch = _fields.Batch(changed: changed);
        ((IJSInProcessRuntime)JS).InvokeVoid("performance.mark", "cost-apply");
        var clock = Stopwatch.StartNew();
        _source!.Apply(batch);
        _status = $"Batch {++_batch}: apply {clock.Elapsed.TotalMilliseconds.ToString("F3", CultureInfo.InvariantCulture)} ms";
    }
    // [DEBUG-oom-a4f2] Disposable inspection only. Weak references do not keep old graphs alive.
    private readonly Dictionary<string, List<WeakReference<object>>> _tracked = new();
    private static object? Field(object? target, string name) => target?.GetType().GetField(name, BindingFlags.Instance | BindingFlags.NonPublic | BindingFlags.Public)?.GetValue(target);
    private static object? Property(object? target, string name) => target?.GetType().GetProperty(name, BindingFlags.Instance | BindingFlags.NonPublic | BindingFlags.Public)?.GetValue(target);
    private void Track(string kind, object? target)
    {
        if (target is null) return;
        if (!_tracked.TryGetValue(kind, out var list)) _tracked[kind] = list = [];
        if (!list.Any(w => w.TryGetTarget(out var prior) && ReferenceEquals(prior, target))) list.Add(new(target));
    }
    private void Probe()
    {
        var before = GC.GetTotalMemory(false);
        if (Intervention == "drop-paints" && Field(Field(_pivot, "_grid"), "_paints") is IList paints) paints.Clear();
        if (ForceGc) { GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect(); }
        var managed = GC.GetTotalMemory(false);
        var report = Field(_pivot, "_report") as PivotReport;
        Track("Report", report); Track("Cube", report?.Cube); Track("Snapshot", _source?.Snapshot); Track("Answer", Field(_source, "_heldAnswer"));
        var history = Field(_pivot, "_history");
        var versions = (Field(history, "_versions") as IEnumerable)?.Cast<object>().ToArray() ?? [];
        var heldPaints = (Field(Field(_pivot, "_grid"), "_paints") as IEnumerable)?.Cast<object>().ToArray() ?? [];
        var paintRows = heldPaints.SelectMany(p => (Property(p, "Rows") as IEnumerable)?.Cast<object?>() ?? []).OfType<PivotReportRow>().ToArray();
        var paintReports = paintRows.Select(r => r.Report).Distinct(ReferenceEqualityComparer.Instance).ToArray();
        var sourceVersions = (Field(_source, "_answered") as IEnumerable)?.Cast<object>().ToArray() ?? [];
        Console.WriteLine("[DEBUG-oom-a4f2] " + JsonSerializer.Serialize(new {
            update = _batch, records = _rows.Length, reportRows = report?.Rows.Count, forcedGc = ForceGc, intervention = Intervention,
            managedBeforeGc = before, managedBytes = managed, allocatedBytes = GC.GetTotalAllocatedBytes(false), threadAllocatedBytes = GC.GetAllocatedBytesForCurrentThread(),
            gc0 = GC.CollectionCount(0), gc1 = GC.CollectionCount(1), gc2 = GC.CollectionCount(2),
            tracked = _tracked.ToDictionary(kv => kv.Key, kv => new { created = kv.Value.Count, alive = kv.Value.Count(w => w.TryGetTarget(out _)) }),
            roots = new { historyVersions = versions.Length, historyReports = versions.Select(v => ((PivotReport)Property(v, "Report")!).Cube.SourceVersion).ToArray(),
                paints = heldPaints.Length, paintRowReferences = paintRows.Length, paintDistinctReports = paintReports.Length,
                paintReportVersions = paintReports.Cast<PivotReport>().Select(r => r.Cube.SourceVersion).ToArray(),
                sourceAnsweredSnapshots = sourceVersions.Length }
        }));
    }

}
