#pragma warning disable xUnit1051

using System.Diagnostics;
using System.Globalization;
using Microsoft.AspNetCore.Components;
using System.Reflection;
using System.Text.Json;
using Bunit;
using ExGrid;
using ExGrid.Components;
using ExPivot.Components.Tests.Support;
using ExPivot.Engine;
using Xunit;
using PC = ExPivot.Components.ExPivot;

namespace ExPivot.Components.Tests;

public sealed class LiveFullRenderMeasurement : PivotTestContext
{
    private static readonly PivotSlicing Whole = new() { Budget = TimeSpan.FromDays(1) };
    private const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    private sealed record Row(int Id, string A, string B, decimal Value);
    private static object? Get(object o, string name) => o.GetType().GetField(name, Private)!.GetValue(o);
    private static object? Call(object o, string name, params object?[] args) => o.GetType().GetMethod(name, Private)!.Invoke(o, args);

    [Fact] // ADR-0067/0140: actual ExPivot rendering, observational timing with row-count checks.
    public async Task Actual_ExPivot_render_after_Show()
    {
        SetRendererInfo(new RendererInfo("Server", true));
        var results = new List<object>();
        foreach (var inner in new[] { 10, 100, 400 })
        {
            var records = Enumerable.Range(0, 1000 * inner).Select(i => new Row(i, $"A{i / inner:00000}", $"B{i % inner:0000}", i)).ToArray();
            var fields = PivotFields.Of<Row>().Key("Id", r => r.Id).Text("A", r => r.A).Text("B", r => r.B).Number("Value", r => r.Value);
            var source = PivotSource.From(fields.Build(records), fields.Fields, Whole);
            var layout = new PivotLayout { Rows = [new("A"), new("B")], Values = [new("Value")] };
            var cut = Render<PC>(p => p
                .Add(c => c.Source, source)
                .Add(c => c.Layout, layout)
                .Add(c => c.Culture, CultureInfo.GetCultureInfo("en-US"))
                .Add(c => c.Slicing, Whole)
                .Add(c => c.Caps, new PivotCaps { MaxLeaves = 1_000_000, MaxRows = 1_000_000 })
                .Add(c => c.RedrawInterval, TimeSpan.Zero)
                .Add(c => c.ViewportHeight, (ViewportSize)480)
                .Add(c => c.ViewportWidth, (ViewportSize)1200));
            cut.WaitForAssertion(() => Assert.Equal(1000 * inner + 1001, Grid(cut).Instance.Window.Count), TimeSpan.FromMinutes(1));
            var component = cut.Instance;
            await cut.InvokeAsync(() => Call(component, "StopListening"));
            var initial = (PivotReport)Get(component, "_report")!;
            var query = PivotQuery.For(layout, 1_000_000);
            var paceType = typeof(PC).GetNestedType("Pace", BindingFlags.NonPublic)!;
            var renderMethod = typeof(PC).GetMethod("StateHasChanged", Private)!;
            var runs = new List<object>();
            for (var iteration = -2; iteration < 9; iteration++)
            {
                GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
                Clock.Advance(TimeSpan.FromMilliseconds(1));
                var changed = Enumerable.Range(0, 1000).Select(i => {
                    var at = i * inner;
                    return records[at] = records[at] with { Value = records[at].Value + 1 };
                }).ToArray();
                source.Apply(fields.Batch(changed: changed));
                var answer = await source.AggregateAsync(query);
                var cube = await PivotEngine.CubeAsync(query, answer, source.Fields, Whole);
                // Use the component's own Report/row-comparison/label-width path, before the render timer.
                await cut.InvokeAsync(async () => {
                    var pace = Activator.CreateInstance(paceType, Whole, CancellationToken.None)!;
                    var build = (Task)Call(component, "BuildAsync", cube, layout, pace, CancellationToken.None)!;
                    await build;
                    var built = build.GetType().GetProperty("Result")!.GetValue(build)!;
                    Call(component, "Show", built, layout, (DateTimeOffset?)Clock.GetUtcNow(), true);
                });
                var report = (PivotReport)Get(component, "_report")!;
                var before = cut.FindComponents<ExGridRow<PivotReportRow>>().ToDictionary(x => x.Instance, x => x.RenderCount);
                var grid = Grid(cut);
                var gridBefore = grid.RenderCount;
                var parentBefore = cut.RenderCount;
                var textBefore = RowTexts(cut);
                double elapsed = 0;
                await cut.InvokeAsync(() => {
                    var start = Stopwatch.GetTimestamp();
                    renderMethod.Invoke(component, null);
                    elapsed = Stopwatch.GetElapsedTime(start).TotalMilliseconds;
                    // The timer is valid only if the synchronous renderer reached the actual new Window.
                    Assert.Same(report.Rows, grid.Instance.Window);
                    Assert.True(grid.RenderCount > gridBefore);
                });
                var painted = cut.FindComponents<ExGridRow<PivotReportRow>>().Where(x => !x.Instance.Placeholder).ToArray();
                var textAfter = RowTexts(cut);
                var rendered = painted.Count(x => !before.TryGetValue(x.Instance, out var old) || old != x.RenderCount);
                var mounted = painted.Count(x => !before.ContainsKey(x.Instance));
                var textChanges = textAfter.Zip(textBefore).Count(x => x.First != x.Second);
                Assert.Equal(painted.Length, rendered);
                Assert.Equal(0, mounted);
                Assert.True(textChanges > 0);
                Assert.Same(Get(component, "_cellChangedAt"), grid.Instance.CellChangedAt);
                if (iteration >= 0) runs.Add(new {
                    renderIncludingWindowCheckMs = elapsed,
                    painted = painted.Length, rendered, mounted, paintedTextRowsChanged = textChanges,
                    rowRenderCalls = painted.Sum(x => x.RenderCount - (before.TryGetValue(x.Instance, out var old) ? old : 0)),
                    gridRenderCalls = grid.RenderCount - gridBefore, pivotRenderCalls = cut.RenderCount - parentBefore,
                    columns = grid.Instance.Columns.Count, hasChangeHighlight = grid.Instance.CellChangedAt is not null
                });
            }
            results.Add(new { records = records.Length, reportRows = initial.Rows.Count, changes = 1000,
                rowFields = 2, valueFields = 1, viewportHeight = 480, viewportWidth = 1200,
                scope = "Actual ExPivot StateHasChanged after its BuildAsync and Show; includes child ExGrid Window validation, component render/diff, lazy cells/Change Highlight and bUnit DOM processing; excludes BuildAsync, Show, browser layout and paint.", runs });
            File.WriteAllText(Path.Combine(Environment.GetEnvironmentVariable("LIVE_COST_OUTPUT")!, "pivot-full-render.json"), JsonSerializer.Serialize(results, new JsonSerializerOptions { WriteIndented = true }));
            Console.WriteLine($"Actual ExPivot render completed report rows={initial.Rows.Count}");
            cut.Dispose();
        }
    }
}
