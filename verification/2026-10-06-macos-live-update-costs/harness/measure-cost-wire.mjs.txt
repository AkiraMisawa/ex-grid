import { chromium } from '@playwright/test';
import fs from 'node:fs';
const base=process.env.COST_SERVER;
const browser=await chromium.launch({channel:'chrome',headless:true});
const results=[];
for(const rows of [100000,1000000]) for(const changes of [1,100,1000]) {
 const page=await browser.newPage({viewport:{width:1400,height:1100}});
 const logs=[];
 page.on('console',m=>{if(m.type()==='error'||m.type()==='warning') logs.push({type:m.type(),text:m.text()});});
 page.on('pageerror',e=>logs.push({type:'exception',text:e.message}));
 const cdp=await page.context().newCDPSession(page);await cdp.send('Network.enable');
 const frames=[],protocolHandshakes=[];let handshake;
 const kinds=['JS.RenderBatch','OnRenderCompleted','JS.EndInvokeDotNet','BeginInvokeDotNetFromJS','JS.BeginInvokeJS','JS.Error'];
 // A WebSocket message can contain the JSON handshake and/or several binary SignalR messages.
 // Retain an incomplete message until the next WebSocket event instead of treating that event as a message.
 const pending={in:Buffer.alloc(0),out:Buffer.alloc(0)};let socketFrame=0;
 const take=(dir,event)=>{
  const r=event.response,raw=Buffer.from(r.payloadData,r.opcode===2?'base64':'utf8');
  const frame=++socketFrame,pieces=[];
  if(r.opcode!==1&&r.opcode!==2)return;
  let b=Buffer.concat([pending[dir],raw]),at=0;
  // The SignalR protocol handshake is JSON terminated by Record Separator, even when sent in a binary frame.
  if(b[0]===123){
   const end=b.indexOf(30);
   if(end<0){pending[dir]=b;return;}
   const payload=JSON.parse(b.subarray(0,end).toString('utf8'));
   protocolHandshakes.push({dir,opcode:r.opcode,requestId:event.requestId,firstBytesHex:b.subarray(0,Math.min(end+1,64)).toString('hex'),payload});
   pieces.push(b.subarray(0,end+1));at=end+1;
  }
  while(at<b.length){
   const start=at;let length=0,shift=0,byte,complete=false;
   while(at<b.length){
    if(shift>28)throw new Error('invalid SignalR message length');
    byte=b[at++];length+=(byte&127)*2**shift;shift+=7;
    if(!(byte&128)){complete=true;break;}
   }
   if(!complete||at+length>b.length){at=start;break;}
   pieces.push(b.subarray(start,at+length));at+=length;
  }
  pending[dir]=b.subarray(at);
  for(const part of pieces){const t=part.toString('latin1',0,96);frames.push({dir,bytes:part.length,kind:kinds.find(k=>t.includes(k))??'other',socketFrame:frame,messagesInSocketFrame:pieces.length});}
 };
 cdp.on('Network.webSocketFrameReceived',e=>take('in',e));
 cdp.on('Network.webSocketFrameSent',e=>take('out',e));
 cdp.on('Network.webSocketHandshakeResponseReceived',e=>handshake=e.response.headers);
 const wire=async()=>await (await fetch(`${base}/api/wire`)).json();
 const count=(dir,kind)=>frames.filter(f=>f.dir===dir&&f.kind===kind).length;
 const until=async(condition)=>{const deadline=Date.now()+60000;while(!condition()){if(Date.now()>deadline)throw new Error('circuit did not complete: '+JSON.stringify(frames.slice(-8)));await new Promise(r=>setTimeout(r,2));}};
 await page.goto(`${base}/grid-cost?rows=${rows}&batch=${changes}`);
 await page.locator('.ex-row').first().waitFor({timeout:120000});
 await until(()=>count('out','OnRenderCompleted')>=count('in','JS.RenderBatch'));
 // Correlate the WebSocket handshake with Kestrel counters by its explicit connection ID.
 let oldBatch=count('in','JS.RenderBatch'),oldAck=count('out','OnRenderCompleted'),oldEnd=count('in','JS.EndInvokeDotNet');
 await page.locator('#cost-tick').click();
 await until(()=>count('in','JS.RenderBatch')>oldBatch&&count('out','OnRenderCompleted')>oldAck&&count('in','JS.EndInvokeDotNet')>oldEnd);
 const circuit=handshake?.['X-Bench-Connection']??handshake?.['x-bench-connection'];
 if(!circuit)throw new Error('the WebSocket handshake did not identify its Kestrel connection');
 const runs=[];
 for(let run=-3;run<15;run++) {
  const w0=(await wire()).find(c=>c.id===circuit).written;
  const offset=frames.length;oldBatch=count('in','JS.RenderBatch');oldAck=count('out','OnRenderCompleted');oldEnd=count('in','JS.EndInvokeDotNet');
  await page.locator('#cost-tick').click();
  await until(()=>count('in','JS.RenderBatch')>oldBatch&&count('out','OnRenderCompleted')>oldAck&&count('in','JS.EndInvokeDotNet')>oldEnd);
  const w1=(await wire()).find(c=>c.id===circuit).written;
  const during=frames.slice(offset);
  if(count('in','JS.RenderBatch')-oldBatch!==1||count('out','OnRenderCompleted')-oldAck!==1||count('in','JS.EndInvokeDotNet')-oldEnd!==1)throw new Error('one click did not produce exactly one render, ack and event completion');
  if(pending.in.length||pending.out.length)throw new Error('an incomplete SignalR message remains at update completion');
  if(run>=0)runs.push({batches:count('in','JS.RenderBatch')-oldBatch,acks:count('out','OnRenderCompleted')-oldAck,eventCompletions:count('in','JS.EndInvokeDotNet')-oldEnd,wireBytes:w1-w0,renderPayloadBytes:during.filter(f=>f.dir==='in'&&f.kind==='JS.RenderBatch').reduce((s,f)=>s+f.bytes,0),frames:during});
 }
 results.push({rows,changes,circuitConnection:circuit,protocolHandshakes,paintedRows:await page.locator('.ex-row').count(),webSocketExtensions:handshake?.['Sec-WebSocket-Extensions']??handshake?.['sec-websocket-extensions'],runs,logs});
 fs.writeFileSync(`${process.env.LIVE_COST_OUTPUT}/grid-wire.json`,JSON.stringify({browser:browser.version(),results},null,2));
 console.log(`wire rows=${rows} changes=${changes} complete`);
 if(logs.length)throw new Error(JSON.stringify(logs));
 await page.close();
}
await browser.close();
