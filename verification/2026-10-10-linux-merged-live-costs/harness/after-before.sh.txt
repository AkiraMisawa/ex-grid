#!/usr/bin/env bash
# The code before the fix (4a289b0; its host is the morning's hosts-perf-final) against the fix, alternated,
# in the browser (PV-21) and on CoreCLR (PerfQuestion).
S=/tmp/claude-0/-home-user-ex-grid/745dc86b-9814-555a-832d-a3eebdd9e2d0/scratchpad
P=$S/perf; W=/home/user/ex-grid/.claude/worktrees
L=$S/fix/before; mkdir -p $L
echo "build start $(date +%T)" >> $L/run.log
(cd $W/perf-before && timeout 1800 nix develop /home/user/ex-grid -c dotnet build samples/ExGrid.DemoApi -c Release > $L/api-before.log 2>&1); echo "api before exit $? $(date +%T)" >> $L/run.log
(cd $W/perf-before && timeout 1800 nix develop /home/user/ex-grid -c dotnet build tests/ExPivot.Components -c Release > $L/components-before.log 2>&1); echo "components before exit $? $(date +%T)" >> $L/run.log
(cd $W/perf-before && timeout 120 nix develop /home/user/ex-grid -c dotnet build-server shutdown > /dev/null 2>&1); pkill -f "[M]SBuild.dll /noautoresponse"; sleep 20
B=$W/live-data-best/tests/ExGrid.Browser
REC=$W/live-data-best/verification/2026-10-10-linux
n=0
for side in before fixed before fixed; do
  n=$((n+1)); tag=b$n-$side
  host=$side; [ $side = before ] && host=final
  echo "$tag start $(date +%T) load $(cut -d' ' -f1-3 /proc/loadavg)" >> $L/run.log
  fuser -k 5440/tcp 2>/dev/null; fuser -k -TERM 8440/tcp 2>/dev/null; sleep 3
  (cd $B && setsid node static-host.mjs $P/hosts-perf-$host/wwwroot 5440 > $L/$tag-static.log 2>&1 &)
  rm -rf $P/pv21/api-data-$side; mkdir -p $P/pv21/api-data-$side
  (cd $W/perf-$side/samples/ExGrid.DemoApi/bin/Release/net10.0 && ASPNETCORE_ENVIRONMENT=Development EXGRID_DEMO_TRADES=1000000 \
     EXGRID_DEMO_DATA=$P/pv21/api-data-$side setsid nix develop /home/user/ex-grid -c dotnet ExGrid.DemoApi.dll --urls http://localhost:8440 > $L/$tag-api.log 2>&1 &)
  for i in $(seq 1 150); do
    curl -sf http://localhost:5440/wide >/dev/null && curl -sf http://localhost:8440/api/status >/dev/null && break
    sleep 2
  done
  rm -f $REC/metrics.json
  (cd $B && EXGRID_BASE_URL=http://localhost:5440 EXGRID_MEASURE=pivot EXGRID_MEASURE_CSV=$P/pv21/trades-1000000.csv \
     timeout 3600 xvfb-run -a -s "-screen 0 2560x1600x24" nix develop '/home/user/ex-grid#browser' -c \
     npx playwright test -c playwright.chromium.config.mjs --project=chrome measure-pivot.spec.mjs > $L/$tag-run.log 2>&1)
  echo "$tag end exit $? $(date +%T)" >> $L/run.log
  cp $REC/metrics.json $L/$tag-metrics.json 2>/dev/null
  fuser -k 5440/tcp 2>/dev/null; fuser -k -TERM 8440/tcp 2>/dev/null; sleep 5; fuser -k 8440/tcp 2>/dev/null
done
n=0
for side in before fixed fixed before; do
  n=$((n+1)); out=$L/q$n-$side; mkdir -p $out
  echo "q$n $side start $(date +%T) load $(cut -d' ' -f1-3 /proc/loadavg)" >> $L/run.log
  (cd $W/perf-$side/tests/ExPivot.Components/bin/Release/net10.0 && DOTNET_TieredCompilation=0 PERF_OUT=$out PERF_RUNS=5 \
    timeout 1500 nix develop /home/user/ex-grid -c dotnet ExPivot.Components.Tests.dll -class ExPivot.Components.Tests.PerfQuestion > $out/log 2>&1)
  echo "q$n $side end exit $? $(date +%T)" >> $L/run.log
done
echo "done $(date +%T)" >> $L/run.log
